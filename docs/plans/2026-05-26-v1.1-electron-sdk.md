# v1.1-electron-sdk Implementation Plan

> Part of initiative: `docs/initiatives/glitch-spot/INITIATIVE.md`

> **For agentic workers:** REQUIRED SUB-SKILL: Use `superpowers:subagent-driven-development` (recommended) or `superpowers:executing-plans` to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship `@glitch-spot/electron` — an npm SDK that embeds the Glitch Spot annotation overlay into any Electron+TypeScript+React app via a child BrowserWindow, with hotkey + programmatic + renderer triggers and a single `onSnapshot` callback.

**Architecture:** A hidden overlay `BrowserWindow` is created at `GlitchSpot.init()` time, parented to the host's main window. On trigger, `webContents.capturePage()` snapshots the renderer; the frame is sent to the overlay via IPC; the overlay (the existing React annotation UI, adapted for Electron transport) renders the frozen screenshot; on `Cmd+Return` the overlay generates an annotated PNG via HTML Canvas, then delivers a `GlitchSnapshot` to the host callback. The existing overlay React source is extended with a runtime-detected bridge abstraction so the same code works under both Tauri and Electron.

**Tech Stack:** TypeScript, React 19, Electron ≥28 (peer dep), tsup (SDK build), vitest + jsdom (tests), Vite (overlay bundle build)

// ABOUTME: Runtime-detected IPC bridge for the overlay UI.
// ABOUTME: TauriBridge wraps @tauri-apps/api; ElectronBridge wraps window.glitchSpotBridge.
import { listen, invoke } from '@tauri-apps/api/event'
import type { AnnotationPayload } from '../../packages/electron/src/types'

export interface OverlayBridge {
  onFrame(cb: (blobUrl: string) => void): Promise<() => void>
  getFrame(): Promise<string | null>
  saveSnapshot(mode: string, annotations: AnnotationPayload[]): Promise<void>
  cancel(): void
}

type TauriFrame = {
  width: number; height: number; scale_factor: number; png: number[]; display_index: number
}

function frameToBlob(frame: TauriFrame): string {
  const bytes = new Uint8Array(frame.png)
  const blob = new Blob([bytes], { type: 'image/png' })
  return URL.createObjectURL(blob)
}

class TauriBridge implements OverlayBridge {
  async onFrame(cb: (blobUrl: string) => void): Promise<() => void> {
    return listen<TauriFrame>('glitch-spot://frame', (e) => cb(frameToBlob(e.payload)))
  }
  async getFrame(): Promise<string | null> {
    const frame = await invoke<TauriFrame | null>('get_latest_frame').catch(() => null)
    return frame ? frameToBlob(frame) : null
  }
  async saveSnapshot(mode: string, annotations: AnnotationPayload[]): Promise<void> {
    await invoke('save_snapshot', { mode, annotations })
  }
  cancel(): void {
    invoke('hide_overlay')
  }
}

interface NativeBridge {
  onFrame(cb: (base64Png: string) => void): void
  offFrame(cb: (base64Png: string) => void): void
  saveSnapshot(mode: string, annotations: AnnotationPayload[], cleanBase64: string, annotatedBase64: string): void
  cancel(): void
}

declare global {
  interface Window {
    __TAURI__?: unknown
    glitchSpotBridge?: NativeBridge
  }
}

class ElectronBridge implements OverlayBridge {
  private cleanBase64: string | null = null
  private cleanBlobUrl: string | null = null
  private frameHandler: ((b64: string) => void) | null = null

  async onFrame(cb: (blobUrl: string) => void): Promise<() => void> {
    this.frameHandler = (b64: string) => {
      if (this.cleanBlobUrl) URL.revokeObjectURL(this.cleanBlobUrl)
      this.cleanBase64 = b64
      const bytes = Uint8Array.from(atob(b64), (c) => c.charCodeAt(0))
      const blob = new Blob([bytes], { type: 'image/png' })
      this.cleanBlobUrl = URL.createObjectURL(blob)
      cb(this.cleanBlobUrl)
    }
    window.glitchSpotBridge!.onFrame(this.frameHandler)
    return () => {
      if (this.frameHandler) window.glitchSpotBridge!.offFrame(this.frameHandler)
    }
  }

  async getFrame(): Promise<string | null> {
    return null
  }

  async saveSnapshot(mode: string, annotations: AnnotationPayload[]): Promise<void> {
    if (!this.cleanBase64 || !this.cleanBlobUrl) return
    const { renderAnnotated } = await import('./annotate')
    const annotatedBase64 = await renderAnnotated(
      this.cleanBlobUrl,
      annotations,
      window.devicePixelRatio,
    )
    window.glitchSpotBridge!.saveSnapshot(mode, annotations, this.cleanBase64, annotatedBase64)
  }

  cancel(): void {
    window.glitchSpotBridge!.cancel()
  }
}

export function createBridge(): OverlayBridge {
  if (window.__TAURI__) return new TauriBridge()
  return new ElectronBridge()
}
```

// ABOUTME: Canvas-based annotation renderer used by the Electron overlay path.
// ABOUTME: Draws annotation rectangles and badges onto the clean captured image.
import type { AnnotationPayload } from '../../packages/electron/src/types'

function loadImage(url: string): Promise<HTMLImageElement> {
  return new Promise((resolve, reject) => {
    const img = new Image()
    img.onload = () => resolve(img)
    img.onerror = reject
    img.src = url
  })
}

/**
 * Renders annotation rectangles and numbered badges onto cleanBlobUrl.
 * Returns the result as a raw base64 PNG string (no data: prefix).
 * scaleFactor converts CSS coords to physical pixels (window.devicePixelRatio).
 */
export async function renderAnnotated(
  cleanBlobUrl: string,
  annotations: AnnotationPayload[],
  scaleFactor: number,
): Promise<string> {
  const img = await loadImage(cleanBlobUrl)
  const canvas = document.createElement('canvas')
  canvas.width = img.naturalWidth
  canvas.height = img.naturalHeight
  const ctx = canvas.getContext('2d')!

  ctx.drawImage(img, 0, 0)

  annotations.forEach((rect, idx) => {
    const x = rect.x * scaleFactor
    const y = rect.y * scaleFactor
    const w = rect.w * scaleFactor
    const h = rect.h * scaleFactor

    ctx.strokeStyle = '#f97316'
    ctx.lineWidth = 2 * scaleFactor
    ctx.strokeRect(x, y, w, h)

    const badgeR = 10 * scaleFactor
    const badgeCx = x + badgeR
    const badgeCy = y + badgeR
    ctx.fillStyle = '#f97316'
    ctx.beginPath()
    ctx.arc(badgeCx, badgeCy, badgeR, 0, Math.PI * 2)
    ctx.fill()

    ctx.fillStyle = '#ffffff'
    ctx.font = `bold ${11 * scaleFactor}px sans-serif`
    ctx.textAlign = 'center'
    ctx.textBaseline = 'middle'
    ctx.fillText(String(idx + 1), badgeCx, badgeCy)
    ctx.textAlign = 'left'
    ctx.textBaseline = 'alphabetic'
  })

  return canvas.toDataURL('image/png').split(',')[1]
}
```

// ABOUTME: Uses the OverlayBridge abstraction to work under both Tauri and Electron.
import { useEffect, useReducer, useRef, useState } from 'react'
import { createBridge } from './bridge'
import { DrawingLayer } from './annotations/DrawingLayer'
import { DiscardModal } from './annotations/DiscardModal'
import { initialState, reducer } from './annotations/state'
import { Toolbar } from './Toolbar'
import type { AnnotationPayload } from '../../packages/electron/src/types'

type Mode = 'live' | 'frozen'

const CHECKER_BG =
  "url(\"data:image/svg+xml;utf8,<svg xmlns='http://www.w3.org/2000/svg' width='20' height='20'>" +
  "<rect width='10' height='10' fill='%23999999'/>" +
  "<rect x='10' y='10' width='10' height='10' fill='%23999999'/>" +
  "<rect x='10' y='0' width='10' height='10' fill='%23ffffff'/>" +
  "<rect x='0' y='10' width='10' height='10' fill='%23ffffff'/>" +
  "</svg>\")"

const bridge = createBridge()

export function Overlay() {
  const [mode, setMode] = useState<Mode>('live')
  const [imageUrl, setImageUrl] = useState<string | null>(null)
  const [annotations, dispatch] = useReducer(reducer, initialState)
  const [showDiscardModal, setShowDiscardModal] = useState(false)
  const currentUrlRef = useRef<string | null>(null)

  const handleSave = async () => {
    const payload: AnnotationPayload[] = annotations.rects.map((r) => ({
      id: r.id,
      x: r.x,
      y: r.y,
      w: r.w,
      h: r.h,
      note: r.note,
    }))
    try {
      await bridge.saveSnapshot(mode, payload)
      dispatch({ type: 'CLEAR' })
      setShowDiscardModal(false)
    } catch (err) {
      console.error('saveSnapshot failed', err)
    }
  }

  useEffect(() => {
    ;(window as unknown as { __glitchSpotPresent?: (mode: Mode) => void }).__glitchSpotPresent = (
      newMode,
    ) => {
      if (currentUrlRef.current) URL.revokeObjectURL(currentUrlRef.current)
      currentUrlRef.current = null
      setImageUrl(null)
      setMode(newMode)
      dispatch({ type: 'CLEAR' })
      setShowDiscardModal(false)

    let unlistenFrame: (() => void) | null = null

      setMode('frozen')
      const img = new window.Image()

        setImageUrl(blobUrl)

    bridge.getFrame().then((blobUrl) => {
      if (blobUrl) {
        setMode('frozen')
        currentUrlRef.current = blobUrl
        setImageUrl(blobUrl)
      }
    })

    return () => {
      unlistenFrame?.()
      if (currentUrlRef.current) URL.revokeObjectURL(currentUrlRef.current)
      delete (window as unknown as { __glitchSpotPresent?: (mode: Mode) => void })
        .__glitchSpotPresent
    }
  }, [])

  useEffect(() => {
    const onKey = (e: KeyboardEvent) => {
      if (e.key === 'Escape') {
        e.preventDefault()
        if (annotations.rects.length === 0) {
          bridge.cancel()
        } else {
          setShowDiscardModal(true)
        }
        return
      }
      if ((e.metaKey || e.ctrlKey) && e.key === 'Enter') {
        e.preventDefault()
        if (annotations.rects.length > 0) handleSave()
        return
      }
      if ((e.metaKey || e.ctrlKey) && e.key.toLowerCase() === 'z') {
        e.preventDefault()
        dispatch({ type: 'UNDO' })
        return
      }
    }
    window.addEventListener('keydown', onKey)
    return () => window.removeEventListener('keydown', onKey)
  }, [annotations.rects, mode])

  const backdrop =
    mode === 'frozen' && imageUrl ? `center / 100% 100% no-repeat url(${imageUrl})` : 'transparent'

  return (
    <div style={{ position: 'fixed', inset: 0, background: backdrop }}>
      <div
        style={{
          position: 'absolute',
          inset: 0,
          backgroundImage: CHECKER_BG,
          backgroundRepeat: 'repeat',
          opacity: 0.12,
          pointerEvents: 'none',
        }}
      />
      <DrawingLayer state={annotations} dispatch={dispatch} />
      <Toolbar rectCount={annotations.rects.length} onSave={handleSave} />
      {showDiscardModal && (
        <DiscardModal
          onDiscard={() => {
            setShowDiscardModal(false)
            dispatch({ type: 'CLEAR' })
            bridge.cancel()
          }}
          onSave={() => {
            setShowDiscardModal(false)
            handleSave()
          }}
          onCancel={() => setShowDiscardModal(false)}
        />
      )}
    </div>
  )
}

// ABOUTME: Preload script for the Glitch Spot overlay BrowserWindow.
// ABOUTME: Exposes window.glitchSpotBridge for the overlay React UI via contextBridge.
import { contextBridge, ipcRenderer } from 'electron'
import type { AnnotationPayload } from './types'

type FrameHandler = (base64Png: string) => void
const frameHandlers = new Set<FrameHandler>()

ipcRenderer.on('glitch-spot:frame', (_event, base64Png: string) => {
  frameHandlers.forEach((cb) => cb(base64Png))
})

contextBridge.exposeInMainWorld('glitchSpotBridge', {
  onFrame(cb: FrameHandler): void {
    frameHandlers.add(cb)
  },
  offFrame(cb: FrameHandler): void {
    frameHandlers.delete(cb)
  },
  saveSnapshot(
    mode: string,
    annotations: AnnotationPayload[],
    cleanBase64: string,
    annotatedBase64: string,
  ): void {
    ipcRenderer.send('glitch-spot:snapshot', { mode, annotations, cleanBase64, annotatedBase64 })
  },
  cancel(): void {
    ipcRenderer.send('glitch-spot:cancel')
  },
})

// ABOUTME: Main-process entry point for @glitch-spot/electron.
// ABOUTME: GlitchSpot.init() creates the overlay window and wires IPC + hotkey.
import path from 'node:path'
import {
  BrowserWindow,
  globalShortcut,
  ipcMain,
  screen,
  type App,
} from 'electron'
import type { GlitchSnapshot, GlitchSpotConfig, GlitchSpotInstance, AnnotationPayload } from './types'

export class GlitchSpot {
  private overlay: BrowserWindow
  private mainWindow: BrowserWindow
  private config: Required<GlitchSpotConfig>
  private app: App
  private active = false

  private constructor(app: App, mainWindow: BrowserWindow, config: Required<GlitchSpotConfig>) {
    this.app = app
    this.mainWindow = mainWindow
    this.config = config

    this.overlay = new BrowserWindow({
      parent: mainWindow,
      transparent: true,
      frame: false,
      alwaysOnTop: true,
      skipTaskbar: true,
      show: false,
      webPreferences: {
        preload: path.join(__dirname, 'overlay-preload.cjs'),
        contextIsolation: true,
        nodeIntegration: false,
      },
    })

    this.overlay.loadFile(path.join(__dirname, '..', 'overlay', 'index.html'))

    this.setupIpc()
    this.trackMainWindow()
  }

  private setupIpc(): void {
    ipcMain.on('glitch-spot:trigger', () => this.trigger())

    ipcMain.on('glitch-spot:snapshot', async (_event, payload: {
      mode: string
      annotations: AnnotationPayload[]
      cleanBase64: string
      annotatedBase64: string
    }) => {
      this.overlay.hide()
      this.active = false

      const bounds = this.mainWindow.getContentBounds()
      const snapshot: GlitchSnapshot = {
        id: `snap_${Date.now().toString(16).padStart(12, '0')}`,
        created_at: new Date().toISOString(),
        mode: 'frozen',
        annotations: payload.annotations.map((a) => ({
          id: a.id,
          shape: 'rect',
          coords: { x: a.x, y: a.y, w: a.w, h: a.h },
          note: a.note,
        })),
        context: {
          screen: {
            width: bounds.width,
            height: bounds.height,
            scale_factor: screen.getDisplayMatching(this.mainWindow.getBounds()).scaleFactor,
          },
          os: {
            platform: process.platform,
            version: process.getSystemVersion?.() ?? '',
          },
          app: {
            name: this.app.getName(),
            version: this.app.getVersion(),
            metadata: this.config.metadata(),
          },
        },
        images: {
          clean_base64: payload.cleanBase64,
          annotated_base64: payload.annotatedBase64,
        },
      }

      try {
        await this.config.onSnapshot(snapshot)
      } catch (err) {
        console.error('[glitch-spot] onSnapshot error:', err)
      }
    })

    ipcMain.on('glitch-spot:cancel', () => {
      this.overlay.hide()
      this.active = false
    })
  }

  private trackMainWindow(): void {
    const syncBounds = () => {
      if (this.active || this.overlay.isDestroyed()) return
      const b = this.mainWindow.getContentBounds()
      this.overlay.setBounds(b)
    }
    this.mainWindow.on('resize', syncBounds)
    this.mainWindow.on('move', syncBounds)
  }

  async trigger(): Promise<void> {
    if (this.active) return
    this.active = true

    const nativeImage = await this.mainWindow.webContents.capturePage()
    const base64Png = nativeImage.toPNG().toString('base64')

    const bounds = this.mainWindow.getContentBounds()
    this.overlay.setBounds(bounds)
    this.overlay.show()
    this.overlay.focus()
    this.overlay.webContents.send('glitch-spot:frame', base64Png)
  }

  static init(app: App, mainWindow: BrowserWindow, config: GlitchSpotConfig): GlitchSpotInstance {
    if (!config.onSnapshot) throw new Error('[glitch-spot] onSnapshot is required')

    const fullConfig: Required<GlitchSpotConfig> = {
      hotkey: config.hotkey ?? '',
      onSnapshot: config.onSnapshot,
      metadata: config.metadata ?? (() => ({})),
    }

    const instance = new GlitchSpot(app, mainWindow, fullConfig)

    if (fullConfig.hotkey) {
      const ok = globalShortcut.register(fullConfig.hotkey, () => instance.trigger())
      if (!ok) console.warn(`[glitch-spot] Could not register hotkey: ${fullConfig.hotkey}`)
    }

    return {
      open() { instance.trigger() },
      stop() {
        if (fullConfig.hotkey) globalShortcut.unregister(fullConfig.hotkey)
        ipcMain.removeAllListeners('glitch-spot:snapshot')
        ipcMain.removeAllListeners('glitch-spot:cancel')
        if (!instance.overlay.isDestroyed()) instance.overlay.destroy()
      },
    }
  }
}
