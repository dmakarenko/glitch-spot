# v1.0-tunnel-share Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use `superpowers:subagent-driven-development` (recommended) or `superpowers:executing-plans` to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

> Part of initiative: [`docs/initiatives/glitch-spot/INITIATIVE.md`](../initiatives/glitch-spot/INITIATIVE.md)
> Design spec: [`docs/initiatives/glitch-spot/v1.0-tunnel-share-seed.md`](../initiatives/glitch-spot/v1.0-tunnel-share-seed.md)

**Goal:** Ship a tray-driven "Share latest snapshot…" action that spawns a user-configured tunnel binary (cloudflared / ngrok / Tailscale Funnel) pointing at a new local read-only HTTP server on `127.0.0.1:7821`, and copies a token-protected public URL to the clipboard.

**Architecture:** Two axum listeners — MCP stays on `127.0.0.1:7820` (unchanged), share routes go on a brand-new listener on `127.0.0.1:7821`. Only `7821` gets tunnelled, so the MCP control surface is never exposed. Per-share 128-bit token persisted as `~/.glitch-spot/snapshots/{id}/share-token`. Tunnel lifecycle owned by a new `TunnelManager` state machine that spawns the configured provider on first share and reuses the URL until app quit.

**Tech Stack:** Rust 2021 / axum 0.7 / tokio (multi-thread runtime, already wired) / Tauri 2 / `rand` (`OsRng`) / `subtle` (constant-time compare) / `async-trait` / `toml` / `regex` / `reqwest` (only for ngrok control-API polling, blocking client OK) / `tauri-plugin-notification`.

  ```rust
  use std::io;
  use std::path::Path;

  use base64::engine::general_purpose::URL_SAFE_NO_PAD;
  use base64::Engine;
  use rand::rngs::OsRng;
  use rand::RngCore;
  use subtle::ConstantTimeEq;

  const TOKEN_FILE: &str = "share-token";

  pub fn generate() -> String {
      let mut bytes = [0u8; 16];
      OsRng.fill_bytes(&mut bytes);
      URL_SAFE_NO_PAD.encode(bytes)
  }

  pub fn verify_ct(provided: &str, stored: &str) -> bool {
      // Reject length mismatches explicitly — ConstantTimeEq panics on differing lengths.
      if provided.len() != stored.len() {
          return false;
      }
      provided.as_bytes().ct_eq(stored.as_bytes()).into()
  }

  pub fn write_for(snapshot_dir: &Path, token: &str) -> io::Result<()> {
      let path = snapshot_dir.join(TOKEN_FILE);
      std::fs::write(&path, format!("{token}\n"))?;
      #[cfg(unix)]
      {
          use std::os::unix::fs::PermissionsExt;
          let mut perms = std::fs::metadata(&path)?.permissions();
          perms.set_mode(0o600);
          std::fs::set_permissions(&path, perms)?;
      }
      Ok(())
  }

  pub fn read_for(snapshot_dir: &Path) -> io::Result<Option<String>> {
      let path = snapshot_dir.join(TOKEN_FILE);
      match std::fs::read_to_string(&path) {
          Ok(s) => Ok(Some(s.trim().to_string())),
          Err(e) if e.kind() == io::ErrorKind::NotFound => Ok(None),
          Err(e) => Err(e),
      }
  }
  ```

  ```rust
  pub fn render_share_page(snap: &Snapshot) -> String {

      img {{ max-width: 100%; height: auto; border: 1px solid #ddd; border-radius: 4px; }}
      ol {{ margin-top: 24px; padding-left: 28px; }}
      ol li {{ margin: 8px 0; }}
      footer {{ margin-top: 32px; font-size: 12px; color: #888; }}
      footer a {{ color: #555; }}
    </style>
  </head>
  <body>
    <h1>Glitch Spot snapshot — {id}</h1>
    <img src="annotated.png" alt="annotated screenshot">
    <ol>
  {notes}  </ol>
    <footer>
      <a href="snapshot.json">Open raw JSON</a>
    </footer>
  </body>
  </html>
  "#,
          id = escape_html(&snap.id),
          notes = notes_list,
      )
  }

  fn escape_html(s: &str) -> String {
      let mut out = String::with_capacity(s.len());
      for c in s.chars() {
          match c {
              '&' => out.push_str("&amp;"),
              '<' => out.push_str("&lt;"),
              '>' => out.push_str("&gt;"),
              '"' => out.push_str("&quot;"),
              '\'' => out.push_str("&#39;"),
              c => out.push(c),
          }
      }
      out
  }
  ```

  ```rust
  use std::sync::Arc;

  struct ShareState {
      base: Arc<PathBuf>,
  }

  pub fn build_share_router(base: PathBuf) -> Router {
      let state = ShareState { base: Arc::new(base) };
      Router::new()
          .route("/healthz", get(healthz))
          .route("/share/:id/:token/", get(landing))
          .route("/share/:id/:token/snapshot.json", get(snapshot_json))
          .route("/share/:id/:token/clean.png", get(|s, p| serve_png(s, p, "clean.png")))
          .route("/share/:id/:token/annotated.png", get(|s, p| serve_png(s, p, "annotated.png")))
          .with_state(state)
  }

  async fn healthz() -> &'static str {
      "ok"
  }

  /// Look up the snapshot dir, verify the token, and return its path.
  /// Returns None for any failure condition — 404 is the unified response.
  fn authorize(state: &ShareState, id: &str, provided_token: &str) -> Option<PathBuf> {
      // Guard against path traversal: snapshot ids are `snap_` + 12 hex chars.
      if !is_safe_id(id) {
          return None;
      }
      let dir = state.base.join(id);
      if !dir.is_dir() {
          return None;
      }
      let stored = token::read_for(&dir).ok().flatten()?;
      if !token::verify_ct(provided_token, &stored) {
          return None;
      }
      Some(dir)
  }

  fn is_safe_id(id: &str) -> bool {
      // Match what snapshot::new_id() produces: "snap_" + 12 ascii hex chars.
      // Reject anything else to head off path traversal up front.
      id.len() == 17
          && id.starts_with("snap_")
          && id[5..].chars().all(|c| c.is_ascii_hexdigit())
  }

      State(state): State<ShareState>,
      AxumPath((id, tok)): AxumPath<(String, String)>,
  ) -> Response {

          return StatusCode::NOT_FOUND.into_response();

          Ok(b) => b,
          Err(_) => return StatusCode::NOT_FOUND.into_response(),

      let snap: Snapshot = match serde_json::from_slice(&json_bytes) {
          Ok(s) => s,
          Err(_) => return StatusCode::NOT_FOUND.into_response(),
      };
      let body = html::render_share_page(&snap);
      ([(header::CONTENT_TYPE, "text/html; charset=utf-8")], body).into_response()
  }

  async fn snapshot_json(
      State(state): State<ShareState>,
      AxumPath((id, tok)): AxumPath<(String, String)>,
  ) -> Response {
      let Some(dir) = authorize(&state, &id, &tok) else {
          return StatusCode::NOT_FOUND.into_response();
      };
      match std::fs::read(dir.join("snapshot.json")) {
          Ok(bytes) => (
              [(header::CONTENT_TYPE, "application/json")],
              bytes,
          ).into_response(),
          Err(_) => StatusCode::NOT_FOUND.into_response(),
      }
  }

      State(state): State<ShareState>,
      AxumPath((id, tok)): AxumPath<(String, String)>,

          Err(_) => StatusCode::NOT_FOUND.into_response(),

  }

      }
  ```

  ```rust
      /// Load settings from `{dir}/config.toml`. Missing or malformed files are

      pub fn load_from(dir: &Path) -> io::Result<Self> {
          let path = dir.join(CONFIG_FILE);
          let parsed: ConfigFile = match std::fs::read_to_string(&path) {
              Ok(text) => toml::from_str(&text).unwrap_or_default(),
              Err(e) if e.kind() == io::ErrorKind::NotFound => ConfigFile::default(),
              Err(e) => return Err(e),
          };
          let provider = parsed
              .tunnel
              .provider
              .as_deref()
              .and_then(ProviderKind::from_str)
              .unwrap_or_default();
          Ok(Settings { provider: RwLock::new(provider) })
      }

          *self.provider.read().expect("settings lock poisoned")
      }

      pub fn save_provider(&self, dir: &Path, kind: ProviderKind) -> io::Result<()> {
          {
              let mut w = self.provider.write().expect("settings lock poisoned");
              *w = kind;

          std::fs::write(dir.join(CONFIG_FILE), text)?;
  ```

  ```rust
  #[derive(Debug, thiserror::Error)]
  pub enum ProviderError {
      #[error("{provider} binary not installed (looked for `{binary}` on PATH)")]
      BinaryNotFound { provider: &'static str, binary: &'static str },
      #[error("{provider}: {reason}")]

      /// under a deadline and for killing the child on shutdown.
      async fn spawn(&self, local_port: u16) -> Result<(Child, UrlFuture), ProviderError>;
  ```

  ```rust
                  }
                  .boxed()
              }
          };

      }
  ```

  ```rust
  use std::time::Duration;

  use tokio::process::Child;
  use tokio::sync::{watch, Mutex};

  use crate::share::tunnel::provider::TunnelProvider;

  #[derive(Debug, Clone, PartialEq)]
  pub enum TunnelState {
      Idle,
      Starting,
      Live { url: String },
      Error { reason: String },
  }

  #[derive(Debug, thiserror::Error)]
  pub enum ShareError {
      #[error("tunnel start failed: {0}")]
      TunnelStartFailed(String),
      #[error("no snapshot to share")]
      NoSnapshot,
      #[error("i/o error: {0}")]
      Io(#[from] std::io::Error),
  }

  const DEFAULT_URL_DEADLINE: Duration = Duration::from_secs(10);

  pub struct TunnelManager {
      inner: Mutex<Inner>,
      tx: watch::Sender<TunnelState>,
  }

  struct Inner {
      state: TunnelState,
      child: Option<Child>,
  }

  impl TunnelManager {
      pub fn new() -> Self {
          let (tx, _rx) = watch::channel(TunnelState::Idle);
          TunnelManager {
              inner: Mutex::new(Inner { state: TunnelState::Idle, child: None }),
              tx,
          }
      }

      pub fn current_state(&self) -> TunnelState {
          self.tx.borrow().clone()
      }

      pub fn subscribe(&self) -> watch::Receiver<TunnelState> {
          self.tx.subscribe()
      }

      /// Start the tunnel (if Idle/Error) and return its public URL.
      /// Re-uses the existing URL when already Live.
      pub async fn start_tunnel(
          &self,
          provider: Arc<dyn TunnelProvider>,
          local_port: u16,
      ) -> Result<String, ShareError> {
          self.start_tunnel_with_deadline(provider, local_port, DEFAULT_URL_DEADLINE).await
      }

      pub async fn start_tunnel_with_deadline(
          &self,
          provider: Arc<dyn TunnelProvider>,
          local_port: u16,
          url_deadline: Duration,
      ) -> Result<String, ShareError> {
          // Fast path — already Live.
          {
              let inner = self.inner.lock().await;
              if let TunnelState::Live { url } = &inner.state {
                  return Ok(url.clone());
              }
          }

          self.set_state(TunnelState::Starting).await;

          let spawn_result = provider.spawn(local_port).await;
          let (child, url_future) = match spawn_result {
              Ok(pair) => pair,
              Err(e) => {
                  let reason = e.to_string();
                  self.set_state(TunnelState::Error { reason: reason.clone() }).await;
                  return Err(ShareError::TunnelStartFailed(reason));
              }
          };

          // Stash the child immediately so a concurrent shutdown can kill it.
          {
              let mut inner = self.inner.lock().await;
              inner.child = Some(child);
          }

          let url = match tokio::time::timeout(url_deadline, url_future).await {
              Ok(Ok(u)) => u,
              Ok(Err(reason)) => {
                  self.kill_child().await;
                  self.set_state(TunnelState::Error { reason: reason.clone() }).await;
                  return Err(ShareError::TunnelStartFailed(reason));
              }
              Err(_elapsed) => {
                  let reason = format!(
                      "{} didn't publish a URL within {}s",
                      provider.kind().as_str(),
                      url_deadline.as_secs()
                  );
                  self.kill_child().await;
                  self.set_state(TunnelState::Error { reason: reason.clone() }).await;
                  return Err(ShareError::TunnelStartFailed(reason));
              }
          };

          self.set_state(TunnelState::Live { url: url.clone() }).await;
          Ok(url)
      }

      pub async fn shutdown(&self) {
          self.kill_child().await;
          self.set_state(TunnelState::Idle).await;
      }

      async fn kill_child(&self) {
          let child_opt = {
              let mut inner = self.inner.lock().await;
              inner.child.take()
          };
          if let Some(mut child) = child_opt {
              // SIGTERM (kill() on Unix sends SIGKILL via tokio — fine for our purposes,
              // but we attempt a graceful shutdown first via start_kill, then wait briefly).
              let _ = child.start_kill();
              let _ = tokio::time::timeout(Duration::from_secs(2), child.wait()).await;
          }
      }

      async fn set_state(&self, next: TunnelState) {
          let mut inner = self.inner.lock().await;
          inner.state = next.clone();
          let _ = self.tx.send(next);
      }
  }

  impl Default for TunnelManager {
      fn default() -> Self { Self::new() }
  }
  ```

  ```rust
  use std::process::Stdio;
  use std::time::Duration;

  use async_trait::async_trait;
  use futures::future::FutureExt;
  use serde::Deserialize;
  use tokio::process::{Child, Command};

  use crate::share::tunnel::provider::{ProviderError, ProviderKind, TunnelProvider, UrlFuture};

  pub struct Ngrok;

  #[derive(Debug, Deserialize)]
  struct TunnelsResp {
      tunnels: Vec<TunnelEntry>,
  }

  #[derive(Debug, Deserialize)]
  struct TunnelEntry {
      #[serde(default)]
      proto: String,
      #[serde(default)]
      public_url: String,
  }

  pub fn parse_public_url(body: &str) -> Option<String> {
      let parsed: TunnelsResp = serde_json::from_str(body).ok()?;
      parsed
          .tunnels
          .into_iter()
          .find(|t| t.proto == "https" && !t.public_url.is_empty())
          .map(|t| t.public_url)
  }

  async fn poll_for_url(deadline: Duration) -> Result<String, String> {
      let start = std::time::Instant::now();
      let client = reqwest::Client::builder()
          .timeout(Duration::from_secs(1))
          .build()
          .map_err(|e| format!("reqwest client init failed: {e}"))?;
      while start.elapsed() < deadline {
          match client.get("http://127.0.0.1:4040/api/tunnels").send().await {
              Ok(resp) if resp.status().is_success() => {
                  if let Ok(body) = resp.text().await {
                      if let Some(url) = parse_public_url(&body) {
                          return Ok(url);
                      }
                  }
              }
              _ => {}
          }
          tokio::time::sleep(Duration::from_millis(250)).await;
      }
      Err("ngrok control API did not expose a public URL in time".to_string())
  }

  #[async_trait]
  impl TunnelProvider for Ngrok {
      async fn spawn(&self, local_port: u16) -> Result<(Child, UrlFuture), ProviderError> {
          let child = Command::new("ngrok")
              .arg("http")
              .arg(local_port.to_string())
              .arg("--log=stdout")
              .arg("--log-format=json")
              .stdout(Stdio::piped())
              .stderr(Stdio::piped())
              .kill_on_drop(true)
              .spawn()
              .map_err(|e| {
                  if e.kind() == std::io::ErrorKind::NotFound {
                      ProviderError::BinaryNotFound { provider: "ngrok", binary: "ngrok" }
                  } else {
                      ProviderError::Spawn { provider: "ngrok", reason: e.to_string() }
                  }
              })?;

          let url_future: UrlFuture = async move {
              // Deadline here is intentionally generous (8s); TunnelManager wraps
              // the whole thing in its own DEFAULT_URL_DEADLINE (10s) anyway.
              poll_for_url(Duration::from_secs(8)).await
          }
          .boxed();
          Ok((child, url_future))
      }

      fn kind(&self) -> ProviderKind { ProviderKind::Ngrok }
  }
  ```

  ```rust
  use std::time::Duration;

  use async_trait::async_trait;
  use futures::future::FutureExt;
  use serde::Deserialize;
  use tokio::process::{Child, Command};

  use crate::share::tunnel::provider::{ProviderError, ProviderKind, TunnelProvider, UrlFuture};

  pub struct Tailscale;

  #[derive(Debug, Deserialize)]
  struct StatusRoot {
      #[serde(rename = "Self")]
      self_node: Option<SelfNode>,
  }

  #[derive(Debug, Deserialize)]

      let root: StatusRoot = serde_json::from_str(status_json).ok()?;
      let s = root.self_node?;
      let trimmed = s.dns_name.trim_end_matches('.');
      if trimmed.is_empty() {

      if !out.status.success() {
          return Err(String::from_utf8_lossy(&out.stderr).trim().to_string());
      }
      Ok(String::from_utf8_lossy(&out.stdout).to_string())
  }

  #[async_trait]
  impl TunnelProvider for Tailscale {
      async fn spawn(&self, local_port: u16) -> Result<(Child, UrlFuture), ProviderError> {
          // `tailscale funnel --bg` returns immediately after enabling the funnel.
          // We treat this child as a no-op marker; the actual public surface is
          // managed by the tailscaled daemon already running on the system.
          let child = Command::new("tailscale")
              .arg("funnel")
              .arg("--bg")
              .arg(local_port.to_string())
              .stdout(Stdio::piped())
              .stderr(Stdio::piped())
              .kill_on_drop(true)
              .spawn()

                      ProviderError::BinaryNotFound { provider: "tailscale", binary: "tailscale" }
                  } else {
                      ProviderError::Spawn { provider: "tailscale", reason: e.to_string() }
                  }
              })?;

          let url_future: UrlFuture = async move {
              // Allow tailscaled a couple of seconds to register the funnel.
              tokio::time::sleep(Duration::from_millis(500)).await;
              let status = fetch_tailscale_status().await?;
              parse_funnel_url(&status)
                  .ok_or_else(|| "tailscale status: no Self.DNSName".to_string())
          }
          .boxed();

          Ok((child, url_future))
      }

      fn kind(&self) -> ProviderKind { ProviderKind::Tailscale }
  }
  ```

  ```rust
  // ABOUTME: End-to-end integration test for the share flow using MockTunnelProvider.
  // ABOUTME: Writes a real snapshot to disk, then hits the share Router via tower::oneshot.

  use std::sync::Arc;

  use axum::body::Body;
  use axum::http::{Method, Request, StatusCode};
  use glitch_spot::share::{router::build_share_router, token, tunnel::mock::MockTunnelProvider, tunnel::TunnelManager};
  use tower::ServiceExt;

  fn write_real_snapshot(base: &std::path::Path, id: &str) -> std::path::PathBuf {
      let dir = base.join(id);
      std::fs::create_dir_all(&dir).unwrap();
      let snap = serde_json::json!({
          "id": id,
          "created_at": "2026-05-26T12:00:00Z",
          "mode": "frozen",
          "annotations": [{
              "id": "a1", "shape": "rect",
              "coords": { "x": 0.0, "y": 0.0, "w": 10.0, "h": 10.0 },
              "note": "the cancel button is misaligned by 3 px"
          }],
          "context": {
              "screen": { "width": 1440, "height": 900, "scale_factor": 2.0, "display_index": 0 },
              "os": { "platform": "macos", "version": "26.0" }
          },
          "images": { "clean": "x", "annotated": "y" }
      });
      std::fs::write(dir.join("snapshot.json"), serde_json::to_vec_pretty(&snap).unwrap()).unwrap();
      std::fs::write(dir.join("clean.png"), b"\x89PNG-fixture-clean").unwrap();
      std::fs::write(dir.join("annotated.png"), b"\x89PNG-fixture-annotated").unwrap();
      dir
  }

  #[tokio::test]
  async fn share_flow_yields_html_view_with_note() {
      let temp = tempfile::TempDir::new().unwrap();
      let snap_id = "snap_abcdef012345";
      let snap_dir = write_real_snapshot(temp.path(), snap_id);

      // Mock the tunnel.
      let mgr = TunnelManager::new();
      let provider = Arc::new(MockTunnelProvider::with_url("https://e2e.trycloudflare.com"));
      let tunnel_url = mgr.start_tunnel(provider, 7821).await.unwrap();
      assert_eq!(tunnel_url, "https://e2e.trycloudflare.com");

      // Generate token.
      let tok = token::generate();
      token::write_for(&snap_dir, &tok).unwrap();

      // Hit the share router directly (no real network).
      let app = build_share_router(temp.path().to_path_buf());
      let landing_path = format!("/share/{snap_id}/{tok}/");
      let req = Request::builder().method(Method::GET).uri(&landing_path).body(Body::empty()).unwrap();
      let resp = app.oneshot(req).await.unwrap();

      assert_eq!(resp.status(), StatusCode::OK);
      let body_bytes = axum::body::to_bytes(resp.into_body(), usize::MAX).await.unwrap();
      let body = std::str::from_utf8(&body_bytes).unwrap();
      assert!(body.contains("the cancel button is misaligned by 3 px"));
      assert!(body.contains(r#"<img src="annotated.png""#));

      mgr.shutdown().await;
  }
  ```
